// ACRESCENTAR dentro de match /databases/{database}/documents
// Não substitua suas regras existentes por este arquivo.
match /publicBriefings/{token} {
  allow get: if true;
  allow list: if false;
  allow create: if request.auth != null;
  allow update: if
    request.resource.data.diff(resource.data).affectedKeys()
      .hasOnly(['answers', 'customQuestions', 'status', 'submittedAt', 'updatedAt'])
    && request.resource.data.status == 'completed';
  allow delete: if request.auth != null;
}
